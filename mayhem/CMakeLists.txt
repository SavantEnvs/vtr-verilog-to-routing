# Standalone CMake project for the Mayhem fuzz/test builds of VTR's parser libraries.
# Upstream's top-level CMake needs git submodules (libsdcparse, catch2, ...) that a plain checkout
# lacks, so this builds just the self-contained libraries from their upstream sources.
cmake_minimum_required(VERSION 3.16)
project(vtr_mayhem C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VTR ${CMAKE_CURRENT_SOURCE_DIR}/..)
add_definitions(-DVTR_ASSERT_LEVEL=2)
option(MAYHEM_FUZZ "build the fuzz harnesses" ON)
set(MAYHEM_LSAN_OFF ${CMAKE_CURRENT_SOURCE_DIR}/lsan_off.cc CACHE FILEPATH "LSan-off hook linked into every fuzz binary")

file(GLOB LOG_SRC ${VTR}/libs/liblog/src/*.cpp)
list(FILTER LOG_SRC EXCLUDE REGEX ".*/main\\.cpp$")
add_library(vtrlog STATIC ${LOG_SRC})
target_include_directories(vtrlog PUBLIC ${VTR}/libs/liblog/src)

# libvtrutil (+ generated vtr_version.cpp)
foreach(v VTR_VERSION_MAJOR VTR_VERSION_MINOR VTR_VERSION_PATCH)
  set(${v} 0)
endforeach()
set(VTR_VERSION "0.0.0")
set(VTR_VERSION_SHORT "0.0.0")
set(VTR_VERSION_PRERELEASE "mayhem")
set(VTR_VCS_REVISION "mayhem")
set(VTR_VCS_REVISION_SHORT "mayhem")
set(VTR_COMPILER_INFO "${CMAKE_CXX_COMPILER_ID}")
set(VTR_BUILD_INFO "mayhem")
set(VTR_BUILD_TIMESTAMP "0")
configure_file(${VTR}/libs/libvtrutil/src/vtr_version.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/vtr_version.cpp)
file(GLOB VTRUTIL_SRC ${VTR}/libs/libvtrutil/src/*.cpp)
add_library(vtrutil STATIC ${VTRUTIL_SRC} ${CMAKE_CURRENT_BINARY_DIR}/vtr_version.cpp)
target_include_directories(vtrutil PUBLIC ${VTR}/libs/libvtrutil/src)
target_link_libraries(vtrutil vtrlog)

add_library(pugixml STATIC ${VTR}/libs/EXTERNAL/libpugixml/src/pugixml.cpp)
target_include_directories(pugixml PUBLIC ${VTR}/libs/EXTERNAL/libpugixml/src)
file(GLOB PUGIUTIL_SRC ${VTR}/libs/libpugiutil/src/*.cpp)
add_library(pugiutil STATIC ${PUGIUTIL_SRC})
target_include_directories(pugiutil PUBLIC ${VTR}/libs/libpugiutil/src)
target_link_libraries(pugiutil pugixml vtrutil)

# libarchfpga (capnproto/FPGA-interchange reader disabled: VTR_ENABLE_CAPNPROTO undefined)
file(GLOB ARCH_SRC ${VTR}/libs/libarchfpga/src/*.cpp)
list(FILTER ARCH_SRC EXCLUDE REGEX ".*/main\\.cpp$")
add_library(archfpga STATIC ${ARCH_SRC})
target_include_directories(archfpga PUBLIC ${VTR}/libs/libarchfpga/src)
target_link_libraries(archfpga vtrutil pugixml pugiutil)
add_executable(read_arch ${VTR}/libs/libarchfpga/src/main.cpp)
target_link_libraries(read_arch archfpga)

# librrgraph (RR graph XML reader/writer; capnp binary format disabled)
file(GLOB_RECURSE RR_SRC ${VTR}/libs/librrgraph/src/*/*.cpp)
file(GLOB_RECURSE RR_HDR ${VTR}/libs/librrgraph/src/*/*.h)
set(RR_INC "")
foreach(h ${RR_HDR})
  get_filename_component(d ${h} DIRECTORY)
  list(APPEND RR_INC ${d})
endforeach()
list(REMOVE_DUPLICATES RR_INC)
add_library(rrgraph STATIC ${RR_SRC})
target_include_directories(rrgraph PUBLIC ${RR_INC})
target_link_libraries(rrgraph archfpga vtrutil)

# The fixed architecture the RR graph harness validates inputs against, embedded as a C string.
file(READ ${VTR}/vtr_flow/arch/timing/k4_N4_90nm.xml RR_ARCH_XML)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/fuzz_rr_graph_arch.h.in
     "static const char kArchXml[] = R\"VTRARCH(@RR_ARCH_XML@)VTRARCH\";\n")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/fuzz_rr_graph_arch.h.in
               ${CMAKE_CURRENT_BINARY_DIR}/fuzz_rr_graph_arch.h @ONLY)

# vpr server-mode telegram protocol (vpr/src/server: framing, JSON field parser, options; zlib)
find_package(ZLIB REQUIRED)
set(SRV ${VTR}/vpr/src/server)
add_library(vprserver STATIC ${SRV}/telegrambuffer.cpp ${SRV}/telegramheader.cpp ${SRV}/telegramparser.cpp
            ${SRV}/telegramoptions.cpp ${SRV}/convertutils.cpp ${SRV}/zlibutils.cpp)
target_include_directories(vprserver PUBLIC ${SRV})
target_link_libraries(vprserver vtrutil ZLIB::ZLIB)

# libblifparse (flex/bison)
find_package(BISON REQUIRED 3.3)
find_package(FLEX REQUIRED)
set(BP ${VTR}/libs/EXTERNAL/libblifparse/src)
flex_target(BlifLexer ${BP}/blif_lexer.l ${CMAKE_CURRENT_BINARY_DIR}/blif_lexer.gen.cpp
            COMPILE_FLAGS --header-file=${CMAKE_CURRENT_BINARY_DIR}/blif_lexer.gen.hpp)
bison_target(BlifParser ${BP}/blif_parser.y ${CMAKE_CURRENT_BINARY_DIR}/blif_parser.gen.cpp)
add_flex_bison_dependency(BlifLexer BlifParser)
file(GLOB BLIF_SRC ${BP}/blif*.cpp)
add_library(blifparse STATIC ${BLIF_SRC} ${FLEX_BlifLexer_OUTPUTS} ${BISON_BlifParser_OUTPUT_SOURCE})
target_include_directories(blifparse PUBLIC ${BP} ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(blifparse PRIVATE YY_READ_BUF_SIZE=1048576)
add_executable(blifparse_test ${BP}/main.cpp)
target_link_libraries(blifparse_test blifparse)

if(MAYHEM_FUZZ)
  add_library(lsan_off OBJECT ${MAYHEM_LSAN_OFF})
  add_library(standalone_main OBJECT $ENV{STANDALONE_FUZZ_MAIN})
  foreach(h fuzz_read_arch fuzz_blifparse fuzz_expr_eval fuzz_rr_graph fuzz_server_telegram)
    add_executable(${h} ${CMAKE_CURRENT_SOURCE_DIR}/harnesses/${h}.cpp $<TARGET_OBJECTS:lsan_off>)
    target_link_options(${h} PRIVATE $ENV{LIB_FUZZING_ENGINE})
    add_executable(${h}-standalone ${CMAKE_CURRENT_SOURCE_DIR}/harnesses/${h}.cpp
                   $<TARGET_OBJECTS:lsan_off> $<TARGET_OBJECTS:standalone_main>)
    foreach(t ${h} ${h}-standalone)
      target_link_libraries(${t} rrgraph archfpga blifparse vprserver vtrutil)
      target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    endforeach()
  endforeach()
endif()

# Seed generator for fuzz_rr_graph (not a fuzz target): writes the RR graph skeleton of its device.
add_executable(rr_seed_gen ${CMAKE_CURRENT_SOURCE_DIR}/harnesses/fuzz_rr_graph.cpp)
target_compile_definitions(rr_seed_gen PRIVATE MAYHEM_RR_SEED_GEN)
target_include_directories(rr_seed_gen PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(rr_seed_gen rrgraph archfpga vtrutil)

add_executable(telegram_seed_gen ${CMAKE_CURRENT_SOURCE_DIR}/harnesses/fuzz_server_telegram.cpp)
target_compile_definitions(telegram_seed_gen PRIVATE MAYHEM_TELEGRAM_SEED_GEN)
target_link_libraries(telegram_seed_gen vprserver vtrutil)
